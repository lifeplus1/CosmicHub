### Sample Code Snippet: Geometric Pattern Recognition

Below is a modular TypeScript function for detecting sacred geometric ratios in astrological data,
integrable into `apps/astro/src/utils`:

```typescript
// apps/astro/src/utils/sacredGeometry.ts
import { AstrologyData } from '../types/astrology.types';
import { validateNumber } from '../utils/validation';
import { logCalculation } from '../utils/componentLogger';

export const detectGoldenRatio = (data: AstrologyData): number[] => {
  const validatedAngles = data.planets
    .map(planet => validateNumber(planet.angle, 'Planet angle'))
    .filter(angle => angle !== null) as number[];

  const goldenRatio = 1.61803398875;
  const ratios: number[] = [];

  for (let i = 0; i < validatedAngles.length; i++) {
    for (let j = i + 1; j < validatedAngles.length; j++) {
      const ratio = Math.abs(validatedAngles[i] / validatedAngles[j]);
      if (Math.abs(ratio - goldenRatio) < 0.01) {
        ratios.push(ratio);
        logCalculation(
          `Golden ratio detected: ${ratio} between angles ${validatedAngles[i]} and ${validatedAngles[j]}`
        );
      }
    }
  }

  return ratios;
};
```
